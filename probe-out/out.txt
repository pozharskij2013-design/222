P> P> #!/bin/sh
set -eu

mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
mkdir -p /dev/pts
mount -t devpts devpts /dev/pts
mkdir -p /run /tmp
chmod 1777 /tmp
# cpio deliberately normalizes the build tree to root:root.  Give the player a
# private writable staging directory only after the guest has booted.
chown user:user /home/user
chmod 700 /home/user

insmod /lib/modules/ghostvector_lab.ko
chmod 666 /dev/ghostvector

flag_source=/sys/firmware/qemu_fw_cfg/by_name/opt/ctfcup/flag/raw
password_hash_source=/sys/firmware/qemu_fw_cfg/by_name/opt/ctfcup/password_hash/raw
test_mode=0
case "$(cat /proc/cmdline)" in
    *gv.selftest=1*) test_mode=1 ;;
esac

if [ -r "$flag_source" ]; then
    cat "$flag_source" | /usr/sbin/flagd &
else
    echo 'missing per-instance flag channel' >&2
    poweroff -f
fi

sleep 1
# qemu_fw_cfg exposes raw entries as sysfs binary attributes.  Their reported
# size is zero even when a read returns data, so `test -s` always lies here.
# Check readability, then validate the actual value after reading it.
if [ -r "$password_hash_source" ]; then
    umask 077
    password_hash=$(cat "$password_hash_source")
    case "$password_hash" in
        '$6$'*) ;;
        *) echo 'invalid SSH password hash channel' >&2; poweroff -f ;;
    esac
    printf 'root:*:19000:0:99999:7:::\nuser:%s:19000:0:99999:7:::\n' \
        "$password_hash" > /etc/shadow
    chown root:root /etc/shadow
    chmod 600 /etc/shadow
    /usr/sbin/dropbearkey -t ed25519 -f /run/dropbear_ed25519_host_key >/dev/null
    /usr/sbin/dropbear -F -E -j -k -w \
        -r /run/dropbear_ed25519_host_key -p 22 &
elif [ "$test_mode" -ne 1 ]; then
    echo 'missing SSH password hash channel' >&2
    poweroff -f
fi

if [ "$test_mode" -eq 1 ]; then
    exec /usr/bin/drop-to-ctf /usr/bin/gv-selftest
fi

printf '%s\n' 'GhostVector diagnostic VM: SSH only; serial console is unavailable.'
while :; do
    sleep 3600
done
